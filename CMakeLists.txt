cmake_minimum_required(VERSION 3.22)
project(kalibr2 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(KALIBR2_WITH_ROS2 "Build the native rosbag2 reader" ON)
include(CTest)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(Ceres 2.1 REQUIRED)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs calib3d)
find_package(TBB REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(APRILTAG REQUIRED IMPORTED_TARGET apriltag)

add_library(kalibr2_core src/config.cpp src/directory_source.cpp src/detector.cpp src/solver.cpp src/pipeline.cpp)
target_include_directories(kalibr2_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${OpenCV_INCLUDE_DIRS})
target_link_libraries(kalibr2_core PUBLIC Eigen3::Eigen Ceres::ceres ${OpenCV_LIBS} TBB::tbb yaml-cpp PkgConfig::APRILTAG)
target_compile_options(kalibr2_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(kalibr2 src/main.cpp)
target_link_libraries(kalibr2 PRIVATE kalibr2_core)
if(KALIBR2_WITH_ROS2)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rosbag2_cpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  add_library(kalibr2_rosbag src/rosbag_source.cpp)
  target_link_libraries(kalibr2_rosbag PUBLIC kalibr2_core)
  ament_target_dependencies(kalibr2_rosbag PUBLIC rclcpp rosbag2_cpp sensor_msgs)
  target_compile_options(kalibr2_rosbag PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(kalibr2 PRIVATE kalibr2_rosbag)
  target_compile_definitions(kalibr2 PRIVATE KALIBR2_WITH_ROS2)
endif()
if(BUILD_TESTING)
  foreach(name detector solver pipeline)
    add_executable(test_${name} tests/test_${name}.cpp)
    target_link_libraries(test_${name} PRIVATE kalibr2_core)
    add_test(NAME ${name} COMMAND test_${name})
  endforeach()
  set_tests_properties(solver PROPERTIES TIMEOUT 180)
  if(KALIBR2_WITH_ROS2)
    add_executable(test_rosbag tests/test_rosbag.cpp)
    target_link_libraries(test_rosbag PRIVATE kalibr2_rosbag)
    add_test(NAME rosbag COMMAND test_rosbag)
  endif()
  add_test(NAME cli_help COMMAND kalibr2 --help)
  add_test(NAME cli_end_to_end COMMAND ${CMAKE_COMMAND}
    -DCLI=$<TARGET_FILE:kalibr2> -DGENERATOR=$<TARGET_FILE:test_solver>
    -DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}/fixtures
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.cmake)
  set_tests_properties(cli_end_to_end PROPERTIES TIMEOUT 180)
endif()
install(TARGETS kalibr2 RUNTIME DESTINATION bin)
install(DIRECTORY config DESTINATION share/kalibr2)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION share/kalibr2)
if(KALIBR2_WITH_ROS2)
  install(TARGETS kalibr2 RUNTIME DESTINATION lib/${PROJECT_NAME})
  ament_package()
endif()
